    function getTouchDistance(

      touch1,

      touch2

    ) {


      const dx =

        touch1.clientX -

        touch2.clientX;


      const dy =

        touch1.clientY -

        touch2.clientY;

        dx * dx +

        dy * dy

      );


    }

    /* ========================================

       REMOVE WISH STAR

       ======================================== */


    function removeWishStar(

      record,

      duration = 700

    ) {


      if (

        !record ||

        !record.star ||

        record.removing

      ) {

        return;

      }


      record.removing = true;


      const star =

        record.star;


      const material =

        star.material;


      const startOpacity =

        material.opacity;


      const startTime =

        performance.now();



      function fade(now) {


        const progress =

          Math.min(

            (

              now -

              startTime

            ) /

            duration,

            1

          );


        material.opacity =

          startOpacity *

          (1 - progress);



        if (

          progress < 1

        ) {


          requestAnimationFrame(

            fade

          );


        }


        else {


          if (

            star.parent

          ) {


            star.parent.remove(

              star

            );


          }



          if (

            material.map

          ) {


            material.map.dispose();


          }



          material.dispose();


        }


      }



      requestAnimationFrame(

        fade

      );


    }



    /* ========================================

       CREATE WISH STAR

       ======================================== */


    function createWishStar() {


      if (

        !sceneEl ||

        !sceneEl.object3D ||

        !sceneEl.camera

      ) {


        console.warn(

          '360 scene / camera not ready.'

        );


        return;


      }



      const threeCamera =

        sceneEl.camera;


      threeCamera.updateMatrixWorld(

        true

      );



      /* ========================================

         START POSITION

         ======================================== */


      const cameraPosition =

        new THREE.Vector3();


      threeCamera.getWorldPosition(

        cameraPosition

      );



      const viewDirection =

        new THREE.Vector3();


      threeCamera.getWorldDirection(

        viewDirection

      );


      viewDirection.normalize();



      /*

      星星由目前畫面前方較低位置開始。

      */


      const startPosition =

        cameraPosition

          .clone()

          .addScaledVector(

            viewDirection,

            3.5

          );


      startPosition.y -=

        1.15;



      /* ========================================

         MAXIMUM 5 STARS

         ======================================== */


      let slotIndex;



      if (

        wishStars.length >=

        MAX_WISH_STARS

      ) {


        /*

        移除最舊的一粒。

        */


        const oldest =

          wishStars.shift();



        /*

        新星使用最舊星原本的位置。

        */


        slotIndex =

          oldest.slotIndex;



        removeWishStar(

          oldest,

          700

        );


      }


      else {


        /*

        找目前未使用的位置。

        */


        const usedSlots =

          new Set(

            wishStars.map(

              function(item) {


                return item.slotIndex;


              }

            )

          );



        slotIndex =

          [0, 1, 2, 3, 4]

            .find(

              function(index) {


                return !usedSlots.has(

                  index

                );


              }

            );



        if (

          slotIndex === undefined

        ) {


          slotIndex = 0;


        }


      }



      /* ========================================

         FINAL TARGET

         ======================================== */


      const offset =

        WISH_STAR_OFFSETS[

          slotIndex

        ];



      const targetYaw =

        BIG_STAR_YAW +

        THREE.MathUtils.degToRad(

          offset.yaw

        );



      const targetPitch =

        BIG_STAR_PITCH +

        THREE.MathUtils.degToRad(

          offset.pitch

        );



      const targetDirection =

        new THREE.Vector3(

      function animateRise(now) {


        /*

        停止原本升空動畫。

        ) {

        }

          Math.min(

            (

              now -

              animationStart

            ) /

            RISE_TIME,

            1

          );



        const eased =

          1 -

          Math.pow(

            1 - progress,

            3

          );

          startPosition,

          endPosition,

          eased

        );

          Math.min(

            progress /

            0.18,

            1

          );



        const twinkle =

          0.92 +

          Math.sin(

            now *

            0.006 +

            slotIndex

          ) *

          0.08;

          fadeIn *

          twinkle;

        升空期間慢慢變大。

        */


        const scale =

          1.15 +

          1.25 *

          eased;



        star.scale.set(

          scale,

          scale,

          scale

        );



        if (

          progress < 1

        ) {


          requestAnimationFrame(

            animateRise

          );


        }


        else {
